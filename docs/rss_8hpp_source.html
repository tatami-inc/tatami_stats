68void rss_direct(bool row, const tatami::Matrix<Value_, Index_>& mat, RssBuffers<Output_>& output, const RssOptions<Output_>& opt) {
69 const auto dim = (row ? mat.nrow() : mat.ncol());
70 const auto otherdim = (row ? mat.ncol() : mat.nrow());
71 quickstats::RssOptions<Output_> ropt;
72 ropt.mean_placeholder = opt.mean_placeholder;
73
74 if (mat.sparse()) {
75 tatami::Options topt;
76 topt.sparse_extract_index = false;
77
78 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
79 auto ext = tatami::consecutive_extractor<true>(mat, row, s, l, topt);
80 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(otherdim);
81 quickstats::RssWorkspace<Output_> work;
82 for (Index_ x = 0; x < l; ++x) {
83 auto out = ext->fetch(vbuffer.data(), NULL);
84 const auto res = quickstats::rss(otherdim, out.number, out.value, work, ropt);
85 output.mean[x + s] = res.mean;
86 output.rss[x + s] = res.rss;
87 }
88 }, dim, opt.num_threads);
89
90 } else {
91 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
92 auto ext = tatami::consecutive_extractor<false>(mat, row, s, l);
93 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(otherdim);
94 quickstats::RssWorkspace<Output_> work;
95 for (Index_ x = 0; x < l; ++x) {
96 auto out = ext->fetch(buffer.data());
97 const auto res = quickstats::rss(otherdim, out, work, ropt);
98 output.mean[x + s] = res.mean;
99 output.rss[x + s] = res.rss;
100 }
101 }, dim, opt.num_threads);
102 }

105template<typename Value_, typename Index_, typename Output_>
106void rss_running(bool row, const tatami::Matrix<Value_, Index_>& mat, RssBuffers<Output_>& output, const RssOptions<Output_>& opt) {
107 const auto dim = (row ? mat.nrow() : mat.ncol());
108 const auto otherdim = (row ? mat.ncol() : mat.nrow());
109 if (otherdim == 0) {
110 std::fill_n(output.mean, dim, opt.mean_placeholder);
111 std::fill_n(output.rss, dim, 0);
112 return;
113 }
114
115 assert(opt.num_threads > 0);
116 const bool do_parallel = opt.num_threads > 1;
117 std::optional<std::vector<std::optional<std::vector<Output_> > > > all_partial_mean, all_partial_rss;
118 std::optional<std::vector<Index_> > all_partial_count;
119 if (do_parallel) {
120 // -1, as we'll repurpose the RSS output buffer to store the partial RSS of the first thread.
121 all_partial_rss.emplace(sanisizer::cast<I<decltype(all_partial_rss->size())> >(opt.num_threads - 1));
122 all_partial_mean.emplace(sanisizer::cast<I<decltype(all_partial_mean->size())> >(opt.num_threads));
123 all_partial_count.emplace(sanisizer::cast<I<decltype(all_partial_count->size())> >(opt.num_threads));
124 }
125
126 // We overwrite any existing mean value in the array in the do_parallel=true situation.
127 // So, the initial value doesn't need to be zero.
128 if (!do_parallel) {
129 std::fill_n(output.mean, dim, 0);
130 }
131 std::fill_n(output.rss, dim, 0);
132
133 const bool is_sparse = mat.is_sparse();
134 const int nused = tatami::parallelize([&](int thread, Index_ s, Index_ l) -> void {
135 Output_* rss_ptr;
136 Output_* mean_ptr;
137 std::optional<std::vector<Output_> > cur_rss, cur_mean;
138
139 if (!do_parallel) {
140 // Storing mean and RSS directly in the output vector to cut down two allocations if we're not working in parallel.
141 rss_ptr = output.rss;
142 mean_ptr = output.mean;
143 } else {
144 // Storing the partial RSS directly in the output vector to save ourselves an allocation if we're in the first thread.
145 // We can't do the same for the mean, though, as we need to keep the partial mean and the global mean separate for the reduction.
146 cur_mean.emplace(tatami::cast_Index_to_container_size<std::vector<Output_> >(dim));
147 mean_ptr = cur_mean->data();
148 if (thread == 0) {
149 rss_ptr = output.rss;
150 } else {
151 cur_rss.emplace(tatami::cast_Index_to_container_size<std::vector<Output_> >(dim));
152 rss_ptr = cur_rss->data();
153 }
154 }
155
156 if (is_sparse) {
157 tatami::Options topt;
158 topt.sparse_ordered_index = false;
159 auto ext = tatami::consecutive_extractor<true>(mat, !row, s, l, topt);
160 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(dim);
161 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(dim);
162 auto nonzeros = tatami::create_container_of_Index_size<std::vector<Index_> >(dim);
163
164 for (Index_ x = 0; x < l; ++x) {
165 auto out = ext->fetch(vbuffer.data(), ibuffer.data());
166 AUVEH_NODEP
167 for (Index_ i = 0; i < out.number; ++i) {
168 const auto d = out.index[i];
169 auto& nnz = nonzeros[d];
170 quickstats::update_rss(mean_ptr[d], rss_ptr[d], out.value[i], ++nnz); // increment is safe as 'nnz + 1 <= l' fits in an Index_;
171 }
172 }
173
174 AUVEH_NODEP
175 for (Index_ d = 0; d < dim; ++d) {
176 // otherdim > 0 is guaranteed, so we can use the unsafe version.
177 quickstats::update_rss_with_zeros_unsafe(mean_ptr[d], rss_ptr[d], static_cast<Index_>(l - nonzeros[d]), l);
178 }
179
180 } else {
181 auto ext = tatami::consecutive_extractor<false>(mat, !row, s, l);
182 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(dim);
183
184 for (Index_ x = 0; x < l; ++x) {
185 auto out = ext->fetch(buffer.data());
186 AUVEH_NODEP
187 for (Index_ d = 0; d < dim; ++d) {
188 quickstats::update_rss(mean_ptr[d], rss_ptr[d], out[d], x + 1); // increment is safe as ' x + 1 <= l' fits in an Index_.
189 }
190 }
191 }
192
193 if (do_parallel) {
194 (*all_partial_count)[thread] = l;
195 (*all_partial_mean)[thread] = std::move(cur_mean);
196 if (thread > 0) {
197 (*all_partial_rss)[thread - 1] = std::move(cur_rss);
198 }
199 }
200 }, otherdim, opt.num_threads);
201 assert(nused > 0);
202
203 // Don't check nused > 1, as it's possible for do_parallel = true with nused = 1 if not all threads are used.
204 // This would cause us to leave output.mean and output.rss empty.
205 if (do_parallel) {
206 const auto& ap_count = *all_partial_count;
207 const auto& ap_mean = *all_partial_mean;
208 const auto& ap_rss = *all_partial_rss;
209
210 // Computing the global mean. All ap_count is positive so we don't have to worry about cur_mean[d] being NaN.
211 for (int u = 0; u < nused; ++u) {
212 const Output_ mult = static_cast<Output_>(ap_count[u]) / static_cast<Output_>(otherdim);
213 const auto& cur_mean = *(ap_mean[u]);
214 if (u == 0) {
215 AUVEH_NODEP
216 for (Index_ d = 0; d < dim; ++d) {
217 output.mean[d] = cur_mean[d] * mult;
218 }
219 } else {
220 AUVEH_NODEP
221 for (Index_ d = 0; d < dim; ++d) {
222 output.mean[d] += cur_mean[d] * mult;
223 }
224 }
225 }
226
227 // Combining the RSS. We can use recenter_rss_unsafe() as we are guaranteed that cur_count > 0,
228 // as parallelize() will only ever split into non-empty ranges if those ranges are used.
229 for (int u = 0; u < nused; ++u) {
230 const auto cur_count = ap_count[u];
231 const auto& cur_mean = *(ap_mean[u]);
232 if (u == 0) {
233 AUVEH_NODEP
234 for (Index_ d = 0; d < dim; ++d) {
235 output.rss[d] = quickstats::recenter_rss_unsafe(cur_count, output.rss[d], cur_mean[d], output.mean[d]);
236 }
237 } else {
238 const auto& cur_rss = *(ap_rss[u - 1]);
239 AUVEH_NODEP
240 for (Index_ d = 0; d < dim; ++d) {
241 output.rss[d] += quickstats::recenter_rss_unsafe(cur_count, cur_rss[d], cur_mean[d], output.mean[d]);
242 }
243 }
244 }
245 }
