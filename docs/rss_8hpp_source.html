68void rss_direct(bool row, const tatami::Matrix<Value_, Index_>& mat, RssBuffers<Output_>& output, const RssOptions<Output_>& opt) {
69 const auto dim = (row ? mat.nrow() : mat.ncol());
70 const auto otherdim = (row ? mat.ncol() : mat.nrow());
71 quickstats::RssOptions<Output_> ropt;
72 ropt.mean_placeholder = opt.mean_placeholder;
73
74 if (mat.sparse()) {
75 tatami::Options topt;
76 topt.sparse_extract_index = false;
77
78 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
79 auto ext = tatami::consecutive_extractor<true>(mat, row, s, l, topt);
80 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(otherdim);
81 quickstats::RssWorkspace<Output_> work;
82 for (Index_ x = 0; x < l; ++x) {
83 auto out = ext->fetch(vbuffer.data(), NULL);
84 const auto res = quickstats::rss(otherdim, out.number, out.value, work, ropt);
85 output.mean[x + s] = res.mean;
86 output.rss[x + s] = res.rss;
87 }
88 }, dim, opt.num_threads);
89
90 } else {
91 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
92 auto ext = tatami::consecutive_extractor<false>(mat, row, s, l);
93 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(otherdim);
94 quickstats::RssWorkspace<Output_> work;
95 for (Index_ x = 0; x < l; ++x) {
96 auto out = ext->fetch(buffer.data());
97 const auto res = quickstats::rss(otherdim, out, work, ropt);
98 output.mean[x + s] = res.mean;
99 output.rss[x + s] = res.rss;
100 }
101 }, dim, opt.num_threads);
102 }

105template<typename Value_, typename Index_, typename Output_>
106void rss_running(bool row, const tatami::Matrix<Value_, Index_>& mat, RssBuffers<Output_>& output, const RssOptions<Output_>& opt) {
107 const auto dim = (row ? mat.nrow() : mat.ncol());
108 const auto otherdim = (row ? mat.ncol() : mat.nrow());
109
110 std::fill_n(output.rss, dim, 0);
111 if (otherdim == 0) {
112 std::fill_n(output.mean, dim, opt.mean_placeholder);
113 return;
114 } else {
115 std::fill_n(output.mean, dim, 0);
116 }
117
118 assert(opt.num_threads > 0);
119 const bool do_parallel = opt.num_threads > 1;
120 std::optional<std::vector<std::optional<std::vector<Output_> > > > all_partial_mean, all_partial_rss;
121 std::optional<std::vector<Index_> > all_partial_count;
122 if (do_parallel) {
123 // -1, as we'll repurpose the RSS output buffer to store the partial RSS of the first thread.
124 all_partial_rss.emplace(sanisizer::cast<I<decltype(all_partial_rss->size())> >(opt.num_threads - 1));
125 all_partial_mean.emplace(sanisizer::cast<I<decltype(all_partial_mean->size())> >(opt.num_threads));
126 all_partial_count.emplace(sanisizer::cast<I<decltype(all_partial_count->size())> >(opt.num_threads));
127 }
128
129 const bool is_sparse = mat.is_sparse();
130 const int nused = tatami::parallelize([&](int thread, Index_ s, Index_ l) -> void {
131 Output_* rss_ptr;
132 Output_* mean_ptr;
133 std::optional<std::vector<Output_> > cur_rss, cur_mean;
134
135 if (!do_parallel) {
136 // Storing mean and RSS directly in the output vector to cut down two allocations if we're not working in parallel.
137 rss_ptr = output.rss;
138 mean_ptr = output.mean;
139 } else {
140 // Storing the partial RSS directly in the output vector to save ourselves an allocation if we're in the first thread.
141 // We can't do the same for the mean, though, as we need to keep the partial mean and the global mean separate for the reduction.
142 cur_mean.emplace(tatami::cast_Index_to_container_size<std::vector<Output_> >(dim));
143 mean_ptr = cur_mean->data();
144 if (thread == 0) {
145 rss_ptr = output.rss;
146 } else {
147 cur_rss.emplace(tatami::cast_Index_to_container_size<std::vector<Output_> >(dim));
148 rss_ptr = cur_rss->data();
149 }
150 }
151
152 if (is_sparse) {
153 tatami::Options topt;
154 topt.sparse_ordered_index = false;
155 auto ext = tatami::consecutive_extractor<true>(mat, !row, s, l, topt);
156 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(dim);
157 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(dim);
158 auto nonzeros = tatami::create_container_of_Index_size<std::vector<Index_> >(dim);
159
160 for (Index_ x = 0; x < l; ++x) {
161 auto out = ext->fetch(vbuffer.data(), ibuffer.data());
162 for (Index_ i = 0; i < out.number; ++i) {
163 const auto d = out.index[i];
164 auto& nnz = nonzeros[d];
165 quickstats::update_rss(mean_ptr[d], rss_ptr[d], out.value[i], ++nnz); // increment is safe as 'nnz + 1 <= l' fits in an Index_;
166 }
167 }
168
169 for (Index_ d = 0; d < dim; ++d) {
170 // otherdim > 0 is guaranteed, so we can use the unsafe version.
171 quickstats::update_rss_with_zeros_unsafe(mean_ptr[d], rss_ptr[d], static_cast<Index_>(l - nonzeros[d]), l);
172 }
173
174 } else {
175 auto ext = tatami::consecutive_extractor<false>(mat, !row, s, l);
176 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(dim);
177
178 for (Index_ x = 0; x < l; ++x) {
179 auto out = ext->fetch(buffer.data());
180 for (Index_ d = 0; d < dim; ++d) {
181 quickstats::update_rss(mean_ptr[d], rss_ptr[d], out[d], x + 1); // increment is safe as ' x + 1 <= l' fits in an index.
182 }
183 }
184 }
185
186 if (do_parallel) {
187 (*all_partial_count)[thread] = l;
188 (*all_partial_mean)[thread] = std::move(cur_mean);
189 if (thread > 0) {
190 (*all_partial_rss)[thread - 1] = std::move(cur_rss);
191 }
192 }
193 }, otherdim, opt.num_threads);
194 assert(nused > 0);
195
196 // Don't check nused > 1, as it's possible for do_parallel = true with nused = 1 if not all threads are used.
197 // This would cause us to leave output.mean and output.rss empty.
198 if (do_parallel) {
199 const auto& ap_count = *all_partial_count;
200 const auto& ap_mean = *all_partial_mean;
201 const auto& ap_rss = *all_partial_rss;
202
203 // Computing the global mean. All ap_count is positive so we don't have to worry about cur_mean[d] being NaN.
204 for (int u = 0; u < nused; ++u) {
205 const Output_ mult = static_cast<Output_>(ap_count[u]) / static_cast<Output_>(otherdim);
206 const auto& cur_mean = *(ap_mean[u]);
207 for (Index_ d = 0; d < dim; ++d) {
208 output.mean[d] += cur_mean[d] * mult;
209 }
210 }
211
212 // Combining the RSS. We can use recenter_rss_unsafe() as we are guaranteed that cur_count > 0,
213 // as parallelize() will only ever split into non-empty ranges if those ranges are used.
214 for (int u = 0; u < nused; ++u) {
215 const auto cur_count = ap_count[u];
216 const auto& cur_mean = *(ap_mean[u]);
217 if (u == 0) {
218 for (Index_ d = 0; d < dim; ++d) {
219 output.rss[d] = quickstats::recenter_rss_unsafe(cur_count, output.rss[d], cur_mean[d], output.mean[d]);
220 }
221 } else {
222 const auto& cur_rss = *(ap_rss[u - 1]);
223 for (Index_ d = 0; d < dim; ++d) {
224 output.rss[d] += quickstats::recenter_rss_unsafe(cur_count, cur_rss[d], cur_mean[d], output.mean[d]);
225 }
226 }
227 }
228 }
